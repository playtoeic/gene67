        async function analyzeText() {

            if (rawText === SAMPLE_TEXT) {

                return;

            const loader = document.getElementById('loadingOverlay');
            loader.classList.remove('hidden');

For each phrase, categorize into one of:
- "vp": Verb Phrase (Cụm động từ)
- "np": Noun Phrase (Cụm danh từ / Đại từ)
- "pp": Prepositional Phrase (Cụm giới từ)
- "adjp": Adjective/Adverb Phrase (Cụm tính từ / Phó từ)
- "other": Function words, conjunctions, punctuation (Thành phần lẻ / Từ nối / Trợ động từ)

CRITICAL REQUIREMENTS:
1. Preserve ALL characters, including bracketed Vietnamese notes (e.g., "(thích)").
2. Provide a clear Vietnamese translation for each chunk in 'viTranslation'.
3. Provide the full Vietnamese grammatical name in 'grammarName' (e.g. "Cụm danh từ", "Cụm động từ", "Cụm giới từ", "Cụm tính từ", "Thành phần lẻ").
4. Return ONLY a JSON array.`;

            const payload = {
                contents: [{ parts: [{ text: `Analyze and translate into JSON chunks:\n${rawText}` }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] },
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {

                        items: {
                            type: "OBJECT",
                            properties: {
                                text: { type: "STRING" },
                                type: { type: "STRING", enum: ["vp", "np", "pp", "adjp", "other"] },
                                grammarName: { type: "STRING" },
                                viTranslation: { type: "STRING" }
                            },
                            propertyOrdering: ["text", "type", "grammarName", "viTranslation"]
                        }
                    }
                }
            };

            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

                const response = await fetchWithRetry(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },

        async function fetchWithRetry(url, options, retries = 3, delay = 1000) {
            for (let i = 0; i < retries; i++) {
                try {
                    const res = await fetch(url, options);
                    if (res.ok) return res;
                } catch (e) {
                    if (i === retries - 1) throw e;
                }
                await new Promise(r => setTimeout(r, delay * Math.pow(2, i)));
            }
            return fetch(url, options);
        }
